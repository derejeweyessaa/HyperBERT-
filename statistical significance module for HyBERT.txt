Below is a short research-grade statistical significance module for HyBERT, using both the paired t-test and the Wilcoxon signed-rank test. These tests are commonly used to compare the performance of two machine learning models (e.g., HyBERT vs. CAML) across multiple runs or cross-validation folds.

statistics.py
 
# ==========================================
# statistics.py
# Statistical Significance Testing
# ==========================================

import numpy as np
from scipy.stats import ttest_rel, wilcoxon


class StatisticalTest:

def __init__(self, alpha=0.05):
self.alpha = alpha

# -------------------------------------
# Paired t-test
# -------------------------------------
def paired_ttest(self, baseline, proposed):

t_stat, p_value = ttest_rel(
proposed,
baseline
)

print("\n===== Paired t-test =====")
print(f"T-statistic : {t_stat:.4f}")
print(f"P-value : {p_value:.6f}")

if p_value < self.alpha:
print("Result: Significant Improvement")
else:
print("Result: No Significant Difference")

return t_stat, p_value

# -------------------------------------
# Wilcoxon Signed-Rank Test
# -------------------------------------
def wilcoxon_test(self, baseline, proposed):

stat, p_value = wilcoxon(
proposed,
baseline
)

print("\n===== Wilcoxon Test =====")
print(f"Statistic : {stat:.4f}")
print(f"P-value : {p_value:.6f}")

if p_value < self.alpha:
print("Result: Significant Improvement")
else:
print("Result: No Significant Difference")

return stat, p_value
